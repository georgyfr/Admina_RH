    44→    44→function SortableMission({ mission, onUpdate, onRemove }) {
    45→    45→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: mission.id });
    46→    46→  const style = {
    47→    47→    transform: CSS.Transform.toString(transform),
    48→    48→    transition,
    49→    49→    opacity: isDragging ? 0.5 : 1,
    50→    50→  };
    51→    51→  return (
    52→    52→    <Box ref={setNodeRef} style={style} sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1 }}>
    53→    53→      <IconButton {...attributes} {...listeners} size="small" sx={{ cursor: 'grab', color: 'action.disabled' }}>
    54→    54→        <DragIndicator sx={{ fontSize: 18 }} />
    55→    55→      </IconButton>
    56→    56→      <TextField
    57→    57→        size="small" fullWidth placeholder="Decrire la mission..."
    58→    58→        value={mission.text}
    59→    59→        onChange={(e) => onUpdate(mission.id, e.target.value)}
    60→    60→        sx={{ '& input': { fontSize: '0.85rem' } }}
    61→    61→      />
    62→    62→      <IconButton size="small" onClick={() => onRemove(mission.id)} sx={{ color: '#d32f2f' }}>
    63→    63→        <Cancel sx={{ fontSize: 18 }} />
    64→    64→      </IconButton>
    65→    65→    </Box>
    66→    66→  );
    67→    67→}
