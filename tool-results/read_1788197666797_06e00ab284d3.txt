    51→/* ═══ SORTABLE MISSION ═══ */
    52→function SortableMission({ mission, onUpdate, onRemove }) {
    53→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: mission.id });
    54→  const style = {
    55→    transform: CSS.Transform.toString(transform),
    56→    transition,
    57→    opacity: isDragging ? 0.5 : 1,
    58→  };
    59→  return (
    60→    <Box ref={setNodeRef} style={style} sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1 }}>
    61→      <IconButton {...attributes} {...listeners} size="small" sx={{ cursor: 'grab', color: 'action.disabled' }}>
    62→        <DragIndicator sx={{ fontSize: 18 }} />
    63→      </IconButton>
    64→      <TextField
    65→        size="small" fullWidth placeholder="Decrire la mission..."
    66→        value={mission.text}
    67→        onChange={(e) => onUpdate(mission.id, e.target.value)}
    68→        sx={{ '& input': { fontSize: '0.85rem' } }}
    69→      />
    70→      <IconButton size="small" onClick={() => onRemove(mission.id)} sx={{ color: '#d32f2f' }}>
    71→        <Cancel sx={{ fontSize: 18 }} />
    72→      </IconButton>
    73→    </Box>
    74→  );
    75→}

   134→        const nw = p.missions.findIndex((m) => m.id === over.id);
