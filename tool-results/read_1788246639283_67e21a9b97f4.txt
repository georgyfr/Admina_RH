    26→function useDebounce(value, delay) {
    27→  const [debounced, setDebounced] = useState(value);
    28→  useEffect(() => {
    29→    const t = setTimeout(() => setDebounced(value), delay);
    30→    return () => clearTimeout(t);
    31→  }, [value, delay]);
    32→  return debounced;
    33→}

   210→/* ═══ COMPOSANT QUICK STATUS CHIP MENU ═══ */
   211→function StatusChipMenu({ offre, onStatutChange, loadingId, loadingStatut }) {
   212→  const [anchorEl, setAnchorEl] = useState(null);
   213→  const [confirmAnchor, setConfirmAnchor] = useState(null);
   214→  const [pendingStatut, setPendingStatut] = useState(null);
   215→  const open = Boolean(anchorEl);
   216→  const isLoading = loadingId === offre.id;
   217→  const currentColor = statutOffreColor[offre.statutOffre] || 'default';
   218→
   219→  const handleClick = (e) => { e.stopPropagation(); setAnchorEl(e.currentTarget); };
   220→  const handleClose = () => { setAnchorEl(null); setConfirmAnchor(null); setPendingStatut(null); };
   221→
   222→  const handleSelect = (e, newStatut) => {
   223→    e.stopPropagation();
   224→    setAnchorEl(null);
   225→    if (newStatut === offre.statutOffre) return;
   226→    if (dangerousStatuses.includes(newStatut)) { setPendingStatut(newStatut); setConfirmAnchor(e.currentTarget); return; }
   227→    onStatutChange(offre.id, newStatut);
   228→  };
   229→
   230→  const handleConfirm = (e) => {
   231→    e.stopPropagation(); setConfirmAnchor(null);
   232→    if (pendingStatut) { onStatutChange(offre.id, pendingStatut); setPendingStatut(null); }
   233→  };
   234→  const handleCancelConfirm = (e) => { e.stopPropagation(); setConfirmAnchor(null); setPendingStatut(null); };
   235→
   236→  const statusDotColor = { 'A creer': '#9e9e9e', 'Publiee': '#2e7d32', 'Candidatures en cours': '#f57f17', 'Cloturee': '#616161', 'Annulee': '#d32f2f' };
   237→
   238→  const chipEl = (
   239→    <Chip
   240→      icon={isLoading ? <CircularProgress size={14} sx={{ color: 'inherit', ml: 0.5 }} /> : undefined}
   241→      label={isLoading && loadingStatut ? loadingStatut : offre.statutOffre}
   242→      color={isLoading ? 'default' : currentColor}
   243→      size="small" onClick={handleClick} onDelete={handleClick}
   244→      deleteIcon={<KeyboardArrowDown sx={{ fontSize: 16, color: 'inherit' }} />}
   245→      sx={{ fontWeight: 600, height: 26, fontSize: '0.75rem', cursor: 'pointer', transition: 'all 0.2s' }}
   246→      aria-expanded={open} aria-haspopup="true" role="button" tabIndex={0}
   247→      onKeyDown={e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); handleClick(e); } }}
   248→    />
   249→  );
   250→
   251→  const menuEl = (
   252→    <Menu anchorEl={anchorEl} open={open} onClose={handleClose}
   253→      PaperProps={{ sx: { minWidth: 200, py: 0.5, borderRadius: 2 } }} transitionDuration={150}
   254→    >
   255→      <Typography variant="caption" sx={{ px: 2, py: 0.5, display: 'block', fontWeight: 700, color: 'text.secondary', fontSize: '0.65rem', textTransform: 'uppercase', letterSpacing: 1 }}>Changer le statut</Typography>
   256→      {nomenclatures.statut_offre.map(s => {
   257→        const isActive = s === offre.statutOffre;
   258→        const dot = statusDotColor[s] || '#9e9e9e';
   259→        return (
   260→          <MenuItem key={s} value={s} onClick={(e) => handleSelect(e, s)} selected={isActive} disabled={isActive}
   261→            sx={{ py: 0.8, px: 1.5, gap: 1.5, fontSize: '0.85rem' }}
   262→          >
   263→            <Box sx={{ width: 10, height: 10, borderRadius: '50%', bgcolor: dot, flexShrink: 0 }} />
   264→            {s}
   265→            {isActive && <CheckCircle sx={{ fontSize: 16, ml: 'auto', color: '#1976d2' }} />}
   266→          </MenuItem>
   267→        );
   268→      })}
   269→    </Menu>
   270→  );
   271→
   272→  const confirmMsg = pendingStatut === 'Annulee' ? 'Etes-vous sur de vouloir annuler cette offre ?' : 'Etes-vous sur de vouloir clore cette offre ?';
   273→  const confirmBtnColor = pendingStatut === 'Annulee' ? '#d32f2f' : '#616161';
   274→
   275→  const popoverEl = (
   276→    <Popover anchorEl={confirmAnchor} open={Boolean(confirmAnchor)} onClose={handleCancelConfirm}
   277→      anchorOrigin={{ vertical: 'bottom', horizontal: 'center' }} transformOrigin={{ vertical: 'top', horizontal: 'center' }}
   278→      PaperProps={{ sx: { borderRadius: 2, p: 2, maxWidth: 280 } }} disablePortal
   279→    >
   280→      <Box sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1.5 }}>
   281→        <WarningAmber sx={{ fontSize: 20, color: pendingStatut === 'Annulee' ? '#d32f2f' : '#f57f17' }} />
   282→        <Typography variant="subtitle2" fontWeight="bold" sx={{ fontSize: '0.85rem' }}>Confirmer le changement</Typography>
   283→      </Box>
   284→      <Typography variant="body2" sx={{ fontSize: '0.82rem', color: 'text.secondary', mb: 2, lineHeight: 1.5 }}>{confirmMsg}</Typography>
   285→      <Box sx={{ display: 'flex', gap: 1, justifyContent: 'flex-end' }}>
   286→        <Button size="small" onClick={handleCancelConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem' }}>Annuler</Button>
   287→        <Button size="small" variant="contained" onClick={handleConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem', fontWeight: 600, bgcolor: confirmBtnColor }}>Confirmer</Button>
   288→      </Box>
   289→    </Popover>
   290→  );
   291→
   292→  return <>{chipEl}{menuEl}{popoverEl}</>;
   293→}

   363→function EditableField({ label, value, type = 'text', editKey, editing, onEdit, onChange, options, disabled, endAdornment, multiline }) {
   364→  if (type === 'select') {
   365→    return (

   370→        </Box>
   371→        <FormControl size="small" fullWidth disabled={disabled}>
   372→          <Select
   373→            value={value || ''}
   374→            onChange={e => onChange(editKey, e.target.value)}
   375→            sx={{
   376→              bgcolor: editing === editKey ? '#fffde7' : 'transparent',
   377→              '& .MuiSelect-select': { fontSize: '0.85rem', display: 'flex', alignItems: 'center', gap: 1 },

   381→              <Chip
   382→                label={v}
   383→                size="small"
   384→                color={options?.colorMap?.[v] || 'default'}
   385→                sx={{ fontWeight: 600, height: 26, fontSize: '0.78rem' }}
   386→              />
   387→            )}
   388→          >
   389→            {(options?.list || []).map(opt => (
   390→              <MenuItem key={opt} value={opt} sx={{ fontSize: '0.85rem' }}>{opt}</MenuItem>
   391→            ))}
   392→          </Select>
   393→        </FormControl>
   394→      </Box>
   395→    );
   396→  }
   397→
   398→  if (type === 'date') {
   399→    return (
   400→      <Box sx={{ mb: 2 }}>
   401→        <Box sx={{ display: 'flex', alignItems: 'center', mb: 0.5, gap: 0.5 }}>
   402→          <Typography variant="caption" sx={{ fontSize: '0.7rem', fontWeight: 600, color: 'text.secondary', textTransform: 'uppercase', letterSpacing: 0.5 }}>{label}</Typography>
   403→          {!disabled && <Edit sx={{ fontSize: 13, color: 'text.disabled', cursor: 'pointer', '&:hover': { color: 'primary.main' } }} onClick={() => onEdit(editKey)} />}
   404→        </Box>
   405→        <TextField
   406→          size="small"
   407→          type="date"
   408→          fullWidth
   409→          value={value || ''}
   410→          onChange={e => onChange(editKey, e.target.value)}
   411→          disabled={disabled}
   412→          InputProps={{
   413→            startAdornment: <InputAdornment position="start"><CalendarToday sx={{ fontSize: 16, color: 'action.active' }} /></InputAdornment>,
   414→            sx: { fontSize: '0.85rem' },
   415→          }}
   416→          sx={{
   417→            '& input': { fontSize: '0.85rem' },
   418→            bgcolor: editing === editKey ? '#fffde7' : 'transparent',
   419→            transition: 'background-color 0.2s',
   420→          }}
   421→        />
   422→      </Box>
   423→    );
   424→  }
   425→
   426→  if (type === 'number') {
   427→    return (
   428→      <Box sx={{ mb: 2 }}>
   429→        <Box sx={{ display: 'flex', alignItems: 'center', mb: 0.5, gap: 0.5 }}>
   430→          <Typography variant="caption" sx={{ fontSize: '0.7rem', fontWeight: 600, color: 'text.secondary', textTransform: 'uppercase', letterSpacing: 0.5 }}>{label}</Typography>
   431→          {!disabled && <Edit sx={{ fontSize: 13, color: 'text.disabled', cursor: 'pointer', '&:hover': { color: 'primary.main' } }} onClick={() => onEdit(editKey)} />}
   432→        </Box>
   433→        <TextField
   434→          size="small"
   435→          type="number"
   436→          fullWidth
   437→          value={value ?? ''}
   438→          onChange={e => onChange(editKey, e.target.value ? Number(e.target.value) : '')}
   439→          disabled={disabled}
   440→          InputProps={{
   441→            endAdornment: endAdornment ? <InputAdornment position="end">{endAdornment}</InputAdornment> : null,
   442→            sx: { fontSize: '0.85rem' },
   443→            inputProps: { min: 0 },
   444→          }}
   445→          sx={{
   446→            '& input': { fontSize: '0.85rem' },
   447→            bgcolor: editing === editKey ? '#fffde7' : 'transparent',
   448→            transition: 'background-color 0.2s',
   449→          }}
   450→        />
   451→      </Box>
   452→    );
   453→  }
   454→
   455→  return (
   456→    <Box sx={{ mb: 2 }}>
   457→      <Box sx={{ display: 'flex', alignItems: 'center', mb: 0.5, gap: 0.5 }}>

   459→        {!disabled && <Edit sx={{ fontSize: 13, color: 'text.disabled', cursor: 'pointer', '&:hover': { color: 'primary.main' } }} onClick={() => onEdit(editKey)} />}
   460→      </Box>
   461→      <TextField
   462→        size="small"
   463→        fullWidth
   464→        value={value || ''}
   465→        onChange={e => onChange(editKey, e.target.value)}
   466→        disabled={disabled}
   467→        multiline={multiline}
   468→        rows={multiline ? 3 : 1}
   469→        sx={{
   470→          '& textarea, & input': { fontSize: '0.85rem' },
   471→          bgcolor: editing === editKey ? '#fffde7' : 'transparent',
   472→          transition: 'background-color 0.2s',
   473→        }}
   474→      />
   475→    </Box>
   476→  );
