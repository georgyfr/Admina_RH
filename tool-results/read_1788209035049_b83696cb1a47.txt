    43→/* ═══ SORTABLE MISSION ═══ */
    44→function SortableMission({ mission, onUpdate, onRemove }) {
    45→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: mission.id });
    46→  const style = {
    47→    transform: CSS.Transform.toString(transform),
    48→    transition,
    49→    opacity: isDragging ? 0.5 : 1,
    50→  };
    51→  return (
    52→    <Box ref={setNodeRef} style={style} sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1 }}>
    53→      <IconButton {...attributes} {...listeners} size="small" sx={{ cursor: 'grab', color: 'action.disabled' }}>
    54→        <DragIndicator sx={{ fontSize: 18 }} />
    55→      </IconButton>
    56→      <TextField
    57→        size="small" fullWidth placeholder="Decrire la mission..."
    58→        value={mission.text}
    59→        onChange={(e) => onUpdate(mission.id, e.target.value)}
    60→        sx={{ '& input': { fontSize: '0.85rem' } }}
    61→      />
    62→      <IconButton size="small" onClick={() => onRemove(mission.id)} sx={{ color: '#d32f2f' }}>
    63→        <Cancel sx={{ fontSize: 18 }} />
    64→      </IconButton>
    65→    </Box>
    66→  );
    67→}

   129→    }
