   118→def _sumproduct_month(range_ref, month_num, value_range=None):
   119→    """SUMPRODUCT by month. If value_range is None, counts."""
   120→    if value_range is None:
   121→        return f"SUMPRODUCT((MONTH({range_ref})={month_num})*({range_ref}<>\"\"))"
   122→    return f"SUMPRODUCT((MONTH({range_ref})={month_num})*({range_range}))"
   123→
   124→
   125→def _countif_month_cond(range_ref, month_ref, month_num, criteria):
   126→    """Count rows where month matches AND another column equals criteria."""
   127→    return f"SUMPRODUCT((MONTH({month_ref})={month_num})*({range_ref}=\"{criteria}\"))"

   130→def build_formula(kpi_name, month_num):
   131→    """Build the KPI formula for a given month (1-12)."""
   132→    ds, de = DATA_START, DATA_END
   133→    m = month_num
   134→    
   135→    if kpi_name == 'Time-to-Hire (jours)':
   136→        # Average of (Date Pourvue - Date Demande) where Date Pourvue is in that month
   137→        numerator = (
   138→            f"SUMPRODUCT((MONTH({R_DEM}!$M${ds}:$M${de})={m})"
   139→            f"*({R_DEM}!$M${ds}:$M${de}<>\"\")"
   140→            f"*({R_DEM}!$M${ds}:$M${de}-{R_DEM}!$C${ds}:$C${de}))"
   141→        )
   142→        denominator = (
   143→            f"MAX(SUMPRODUCT((MONTH({R_DEM}!$M${ds}:$M${de})={m})"
   144→            f"*({R_DEM}!$M${ds}:$M${de}<>\"\")),1)"
   145→        )
   146→        return f"IFERROR({numerator}/{denominator},0)"
   147→    
   148→    elif kpi_name == 'Cost-per-Hire (FCFA)':
   149→        # Sum of all cost components / count of hires (Date Embauche Def.) in that month
   150→        cost_sum = (
   151→            f"SUMPRODUCT((MONTH({R_COUT}!$K${ds}:$K${de})={m})"
   152→            f"*({R_COUT}!$F${ds}:$F${de}+{R_COUT}!$G${ds}:$G${de}"
   153→            f"+{R_COUT}!$H${ds}:$H${de}+{R_COUT}!$I${ds}:$I${de}))"
   154→        )
   155→        hire_count = (
   156→            f"MAX(SUMPRODUCT((MONTH({R_CAN}!$AF${ds}:$AF${de})={m})"
   157→            f"*({R_CAN}!$AF${ds}:$AF${de}<>\"\")),1)"
   158→        )
   159→        return f"IFERROR({cost_sum}/{hire_count},0)"
   160→    
   161→    elif kpi_name == 'Quality-of-Hire (/20)':
   162→        # Average evaluation total score for evaluations in that month
   163→        numerator = (
   164→            f"SUMPRODUCT((MONTH({R_EVAL}!$E${ds}:$E${de})={m})"
   165→            f"*({R_EVAL}!$L${ds}:$L${de}<>\"\"))"
   166→        )
   167→        denominator = (
   168→            f"MAX(SUMPRODUCT((MONTH({R_EVAL}!$E${ds}:$E${de})={m})"
   169→            f"*({R_EVAL}!$L${ds}:$L${de}<>\"\")),1)"
   170→        )
   171→        return f"IFERROR({numerator}/{denominator},0)"
   172→    
   173→    elif kpi_name == 'Taux acceptation offres (%)':
   174→        # Pipeline: Accepte / (Accepte + Refuse) in that month
   175→        accepte = f"SUMPRODUCT((MONTH({R_PIPE}!$G${ds}:$G${de})={m})*({R_PIPE}!$H${ds}:$H${de}=\"Accepte\"))"
   176→        refuse = f"SUMPRODUCT((MONTH({R_PIPE}!$G${ds}:$G${de})={m})*({R_PIPE}!$H${ds}:$H${de}=\"Refuse\"))"
   177→        denom = f"MAX({accepte}+{refuse},1)"
   178→        return f"IFERROR({accepte}/{denom},0)"
   179→    
   180→    elif kpi_name == 'Taux de remplissage (%)':
   181→        # Demandes Pourvue / Total demandes in that month
   182→        pourvue = f"SUMPRODUCT((MONTH({R_DEM}!$C${ds}:$C${de})={m})*({R_DEM}!$L${ds}:$L${de}=\"Pourvue\"))"
   183→        total = f"MAX(SUMPRODUCT((MONTH({R_DEM}!$C${ds}:$C${de})={m})*({R_DEM}!$B${ds}:$B${de}<>\"\")),1)"
   184→        return f"IFERROR({pourvue}/{total},0)"
   185→    
   186→    elif kpi_name == 'Taux de recommandation (%)':
   187→        # Grille: "Embauche recommandee" / Total evaluated in that month
   188→        reco = f"SUMPRODUCT((MONTH({R_EVAL}!$E${ds}:$E${de})={m})*({R_EVAL}!$M${ds}:$M${de}=\"Embauche recommandee\"))"
   189→        total = f"MAX(SUMPRODUCT((MONTH({R_EVAL}!$E${ds}:$E${de})={m})*({R_EVAL}!$B${ds}:$B${de}<>\"\")),1)"
   190→        return f"IFERROR({reco}/{total},0)"
   191→    
   192→    elif kpi_name == 'Nb candidats / embauche':
   193→        # New candidatures / hires in that month
   194→        candidats = f"SUMPRODUCT((MONTH({R_CAN}!$Y${ds}:$Y${de})={m})*({R_CAN}!$B${ds}:$B${de}<>\"\"))"
   195→        hires = f"MAX(SUMPRODUCT((MONTH({R_CAN}!$AF${ds}:$AF${de})={m})*({R_CAN}!$AF${ds}:$AF${de}<>\"\")),1)"
   196→        return f"IFERROR({candidats}/{hires},0)"
   197→    
   198→    elif kpi_name == 'Entretiens / embauche':
   199→        # Entretiens realises / hires in that month
   200→        entretiens = f"SUMPRODUCT((MONTH({R_ENT}!$E${ds}:$E${de})={m})*({R_ENT}!$K${ds}:$K${de}=\"Realise\"))"
   201→        hires = f"MAX(SUMPRODUCT((MONTH({R_CAN}!$AF${ds}:$AF${de})={m})*({R_CAN}!$AF${ds}:$AF${de}<>\"\")),1)"
   202→        return f"IFERROR({entretiens}/{hires},0)"
   203→    
   204→    return '0'
