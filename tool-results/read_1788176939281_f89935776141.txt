    24→function useDebounce(value, delay) {
    25→  const [debounced, setDebounced] = useState(value);
    26→  useEffect(() => {
    27→    const t = setTimeout(() => setDebounced(value), delay);
    28→    return () => clearTimeout(t);
    29→  }, [value, delay]);
    30→  return debounced;
    31→}

   208→/* ═══ COMPOSANT QUICK STATUS CHIP MENU ═══ */
   209→function StatusChipMenu({ offre, onStatutChange, loadingId, loadingStatut }) {
   210→  const [anchorEl, setAnchorEl] = useState(null);
   211→  const [confirmAnchor, setConfirmAnchor] = useState(null);
   212→  const [pendingStatut, setPendingStatut] = useState(null);
   213→  const open = Boolean(anchorEl);
   214→  const isLoading = loadingId === offre.id;
   215→  const currentColor = statutOffreColor[offre.statutOffre] || 'default';
   216→
   217→  const handleClick = (e) => { e.stopPropagation(); setAnchorEl(e.currentTarget); };
   218→  const handleClose = () => { setAnchorEl(null); setConfirmAnchor(null); setPendingStatut(null); };
   219→
   220→  const handleSelect = (e, newStatut) => {
   221→    e.stopPropagation();
   222→    setAnchorEl(null);
   223→    if (newStatut === offre.statutOffre) return;
   224→    if (dangerousStatuses.includes(newStatut)) { setPendingStatut(newStatut); setConfirmAnchor(e.currentTarget); return; }
   225→    onStatutChange(offre.id, newStatut);
   226→  };
   227→
   228→  const handleConfirm = (e) => {
   229→    e.stopPropagation(); setConfirmAnchor(null);
   230→    if (pendingStatut) { onStatutChange(offre.id, pendingStatut); setPendingStatut(null); }
   231→  };
   232→  const handleCancelConfirm = (e) => { e.stopPropagation(); setConfirmAnchor(null); setPendingStatut(null); };
   233→
   234→  const statusDotColor = { 'A creer': '#9e9e9e', 'Publiee': '#2e7d32', 'Candidatures en cours': '#f57f17', 'Cloturee': '#616161', 'Annulee': '#d32f2f' };
   235→
   236→  const chipEl = (
   237→    <Chip
   238→      icon={isLoading ? <CircularProgress size={14} sx={{ color: 'inherit', ml: 0.5 }} /> : undefined}
   239→      label={isLoading && loadingStatut ? loadingStatut : offre.statutOffre}
   240→      color={isLoading ? 'default' : currentColor}
   241→      size="small" onClick={handleClick} onDelete={handleClick}
   242→      deleteIcon={<KeyboardArrowDown sx={{ fontSize: 16, color: 'inherit' }} />}
   243→      sx={{ fontWeight: 600, height: 26, fontSize: '0.75rem', cursor: 'pointer', transition: 'all 0.2s' }}
   244→      aria-expanded={open} aria-haspopup="true" role="button" tabIndex={0}
   245→      onKeyDown={e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); handleClick(e); } }}
   246→    />
   247→  );
   248→
   249→  const menuEl = (
   250→    <Menu anchorEl={anchorEl} open={open} onClose={handleClose}
   251→      PaperProps={{ sx: { minWidth: 200, py: 0.5, borderRadius: 2 } }} transitionDuration={150}
   252→    >
   253→      <Typography variant="caption" sx={{ px: 2, py: 0.5, display: 'block', fontWeight: 700, color: 'text.secondary', fontSize: '0.65rem', textTransform: 'uppercase', letterSpacing: 1 }}>Changer le statut</Typography>
   254→      {nomenclatures.statut_offre.map(s => {
   255→        const isActive = s === offre.statutOffre;
   256→        const dot = statusDotColor[s] || '#9e9e9e';
   257→        return (
   258→          <MenuItem key={s} value={s} onClick={(e) => handleSelect(e, s)} selected={isActive} disabled={isActive}
   259→            sx={{ py: 0.8, px: 1.5, gap: 1.5, fontSize: '0.85rem' }}
   260→          >
   261→            <Box sx={{ width: 10, height: 10, borderRadius: '50%', bgcolor: dot, flexShrink: 0 }} />
   262→            {s}
   263→            {isActive && <CheckCircle sx={{ fontSize: 16, ml: 'auto', color: '#1976d2' }} />}
   264→          </MenuItem>
   265→        );
   266→      })}
   267→    </Menu>
   268→  );
   269→
   270→  const confirmMsg = pendingStatut === 'Annulee' ? 'Etes-vous sur de vouloir annuler cette offre ?' : 'Etes-vous sur de vouloir clore cette offre ?';
   271→  const confirmBtnColor = pendingStatut === 'Annulee' ? '#d32f2f' : '#616161';
   272→
   273→  const popoverEl = (
   274→    <Popover anchorEl={confirmAnchor} open={Boolean(confirmAnchor)} onClose={handleCancelConfirm}
   275→      anchorOrigin={{ vertical: 'bottom', horizontal: 'center' }} transformOrigin={{ vertical: 'top', horizontal: 'center' }}
   276→      PaperProps={{ sx: { borderRadius: 2, p: 2, maxWidth: 280 } }} disablePortal
   277→    >
   278→      <Box sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1.5 }}>
   279→        <WarningAmber sx={{ fontSize: 20, color: pendingStatut === 'Annulee' ? '#d32f2f' : '#f57f17' }} />
   280→        <Typography variant="subtitle2" fontWeight="bold" sx={{ fontSize: '0.85rem' }}>Confirmer le changement</Typography>
   281→      </Box>
   282→      <Typography variant="body2" sx={{ fontSize: '0.82rem', color: 'text.secondary', mb: 2, lineHeight: 1.5 }}>{confirmMsg}</Typography>
   283→      <Box sx={{ display: 'flex', gap: 1, justifyContent: 'flex-end' }}>
   284→        <Button size="small" onClick={handleCancelConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem' }}>Annuler</Button>
   285→        <Button size="small" variant="contained" onClick={handleConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem', fontWeight: 600, bgcolor: confirmBtnColor }}>Confirmer</Button>
   286→      </Box>
   287→    </Popover>
   288→  );
   289→
   290→  return <>{chipEl}{menuEl}{popoverEl}</>;
   291→}
