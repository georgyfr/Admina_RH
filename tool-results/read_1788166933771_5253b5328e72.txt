   196→/* ═══ COMPOSANT QUICK STATUS CHIP MENU ═══ */
   197→function StatusChipMenu({ offre, onStatutChange, loadingId, loadingStatut }) {
   198→  const [anchorEl, setAnchorEl] = useState(null);
   199→  const [confirmAnchor, setConfirmAnchor] = useState(null);
   200→  const [pendingStatut, setPendingStatut] = useState(null);
   201→  const open = Boolean(anchorEl);
   202→  const isLoading = loadingId === offre.id;
   203→  const currentColor = statutOffreColor[offre.statutOffre] || 'default';
   204→
   205→  const handleClick = (e) => { e.stopPropagation(); setAnchorEl(e.currentTarget); };
   206→  const handleClose = () => { setAnchorEl(null); setConfirmAnchor(null); setPendingStatut(null); };
   207→
   208→  const handleSelect = (e, newStatut) => {
   209→    e.stopPropagation();
   210→    setAnchorEl(null);
   211→    if (newStatut === offre.statutOffre) return;
   212→    if (dangerousStatuses.includes(newStatut)) { setPendingStatut(newStatut); setConfirmAnchor(e.currentTarget); return; }
   213→    onStatutChange(offre.id, newStatut);
   214→  };
   215→
   216→  const handleConfirm = (e) => {
   217→    e.stopPropagation(); setConfirmAnchor(null);
   218→    if (pendingStatut) { onStatutChange(offre.id, pendingStatut); setPendingStatut(null); }
   219→  };
   220→  const handleCancelConfirm = (e) => { e.stopPropagation(); setConfirmAnchor(null); setPendingStatut(null); };
   221→
   222→  const statusDotColor = { 'A creer': '#9e9e9e', 'Publiee': '#2e7d32', 'Candidatures en cours': '#f57f17', 'Cloturee': '#616161', 'Annulee': '#d32f2f' };
   223→
   224→  const chipEl = (
   225→    <Chip
   226→      icon={isLoading ? <CircularProgress size={14} sx={{ color: 'inherit', ml: 0.5 }} /> : undefined}
   227→      label={isLoading && loadingStatut ? loadingStatut : offre.statutOffre}
   228→      color={isLoading ? 'default' : currentColor}
   229→      size="small" onClick={handleClick} onDelete={handleClick}
   230→      deleteIcon={<KeyboardArrowDown sx={{ fontSize: 16, color: 'inherit' }} />}
   231→      sx={{ fontWeight: 600, height: 26, fontSize: '0.75rem', cursor: 'pointer', transition: 'all 0.2s' }}
   232→      aria-expanded={open} aria-haspopup="true" role="button" tabIndex={0}
   233→      onKeyDown={e => { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); handleClick(e); } }}
   234→    />
   235→  );
   236→
   237→  const menuEl = (
   238→    <Menu anchorEl={anchorEl} open={open} onClose={handleClose}
   239→      PaperProps={{ sx: { minWidth: 200, py: 0.5, borderRadius: 2 } }} transitionDuration={150}
   240→    >
   241→      <Typography variant="caption" sx={{ px: 2, py: 0.5, display: 'block', fontWeight: 700, color: 'text.secondary', fontSize: '0.65rem', textTransform: 'uppercase', letterSpacing: 1 }}>Changer le statut</Typography>
   242→      {nomenclatures.statut_offre.map(s => {
   243→        const isActive = s === offre.statutOffre;
   244→        const dot = statusDotColor[s] || '#9e9e9e';
   245→        return (
   246→          <MenuItem key={s} value={s} onClick={(e) => handleSelect(e, s)} selected={isActive} disabled={isActive}
   247→            sx={{ py: 0.8, px: 1.5, gap: 1.5, fontSize: '0.85rem' }}
   248→          >
   249→            <Box sx={{ width: 10, height: 10, borderRadius: '50%', bgcolor: dot, flexShrink: 0 }} />
   250→            {s}
   251→            {isActive && <CheckCircle sx={{ fontSize: 16, ml: 'auto', color: '#1976d2' }} />}
   252→          </MenuItem>
   253→        );
   254→      })}
   255→    </Menu>
   256→  );
   257→
   258→  const confirmMsg = pendingStatut === 'Annulee' ? 'Etes-vous sur de vouloir annuler cette offre ?' : 'Etes-vous sur de vouloir clore cette offre ?';
   259→  const confirmBtnColor = pendingStatut === 'Annulee' ? '#d32f2f' : '#616161';
   260→
   261→  const popoverEl = (
   262→    <Popover anchorEl={confirmAnchor} open={Boolean(confirmAnchor)} onClose={handleCancelConfirm}
   263→      anchorOrigin={{ vertical: 'bottom', horizontal: 'center' }} transformOrigin={{ vertical: 'top', horizontal: 'center' }}
   264→      PaperProps={{ sx: { borderRadius: 2, p: 2, maxWidth: 280 } }} disablePortal
   265→    >
   266→      <Box sx={{ display: 'flex', alignItems: 'center', gap: 1, mb: 1.5 }}>
   267→        <WarningAmber sx={{ fontSize: 20, color: pendingStatut === 'Annulee' ? '#d32f2f' : '#f57f17' }} />
   268→        <Typography variant="subtitle2" fontWeight="bold" sx={{ fontSize: '0.85rem' }}>Confirmer le changement</Typography>
   269→      </Box>
   270→      <Typography variant="body2" sx={{ fontSize: '0.82rem', color: 'text.secondary', mb: 2, lineHeight: 1.5 }}>{confirmMsg}</Typography>
   271→      <Box sx={{ display: 'flex', gap: 1, justifyContent: 'flex-end' }}>
   272→        <Button size="small" onClick={handleCancelConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem' }}>Annuler</Button>
   273→        <Button size="small" variant="contained" onClick={handleConfirm} sx={{ textTransform: 'none', fontSize: '0.8rem', fontWeight: 600, bgcolor: confirmBtnColor }}>Confirmer</Button>
   274→      </Box>
   275→    </Popover>
   276→  );
   277→
   278→  return <>{chipEl}{menuEl}{popoverEl}</>;
   279→}

   330→/* ═══ COMPOSANT CHAMP ÉDITABLE ═══ */
   331→function EditableField({ label, value, type = 'text', editKey, editing, onEdit, onChange, options, disabled, endAdornment, multiline }) {
   332→  if (type === 'select') {
   333→    return (
   334→      <Box sx={{ mb: 2 }}>
   335→        <Box sx={{ display: 'flex', alignItems: 'center', mb: 0.5, gap: 0.5 }}>
   336→          <Typography variant="caption" sx={{ fontSize: '0.7rem', fontWeight: 600, color: 'text.secondary', textTransform: 'uppercase', letterSpacing: 0.5 }}>{label}</Typography>
   337→          {!disabled && <Edit sx={{ fontSize: 13, color: 'text.disabled', cursor: 'pointer', '&:hover': { color: 'primary.main' } }} onClick={() => onEdit(editKey)} />}
   338→        </Box>
   339→        <FormControl size="small" fullWidth disabled={disabled}>
   340→          <Select
   341→            value={value || ''}
   342→            onChange={e => onChange(editKey, e.target.value)}

   759→  const filtered = useMemo(() => data.filter(o =>
   760→    (filterDep === 'Tous' || o.departement === filterDep) &&
   761→    (filterStatut === 'Tous' || o.statutOffre === filterStatut) &&
   762→    (filterCanal === 'Tous' || o.canalDiffusion === filterCanal)

   764→
   765→  const handleSaveOffre = useCallback((id, formData) => {

   770→      if (o.statutOffre !== formData.statutOffre) changes.push(`Statut : ${o.statutOffre} \u2192 ${formData.statutOffre}`);
   771→      if (o.priorite !== formData.priorite) changes.push(`Priorit\u00e9 : ${o.priorite} \u2192 ${formData.priorite}`);
   772→      if (o.intitule !== formData.intitule) changes.push('Intitul\u00e9 modifi\u00e9');
   773→      if (o.responsable !== formData.responsable) changes.push(`Responsable : ${o.responsable} \u2192 ${formData.responsable}`);
   774→      if (o.budgetAlloue !== formData.budgetAlloue) changes.push('Budget modifi\u00e9');
   775→      if (o.canalDiffusion !== formData.canalDiffusion) changes.push(`Canal : ${o.canalDiffusion} \u2192 ${formData.canalDiffusion}`);
   776→
   777→      const newHist = [...(o.historique || [])];
   778→      changes.forEach(ch => {
   779→        newHist.push({ date: today, evenement: ch, auteur: 'Utilisateur', type: 'modification' });
   780→      });
   781→      if (changes.length === 0) {
   782→        newHist.push({ date: today, evenement: 'Offre modifi\u00e9e', auteur: 'Utilisateur', type: 'modification' });
   783→      }
   784→
   785→      return {
   786→        ...o,
   787→        ...formData,
   788→        historique: newHist,
   789→      };

   792→    setDrawerOffre(prev => prev && prev.id === id ? { ...prev, ...formData } : prev);
   793→    setSnack({ msg: 'Offre mise \u00e0 jour avec succ\u00e8s', severity: 'success' });

   804→    const today = new Date().toLocaleDateString('fr-FR');
   805→    const oldStatut = offre.statutOffre;

   807→    /* Mise à jour optimiste du state */

   809→      if (o.id !== id) return o;
   810→      return {
   811→        ...o,
   812→        statutOffre: newStatut,
   813→        historique: [...(o.historique || []), {
   814→          date: today,
   815→          evenement: `Statut chang\u00e9 : ${oldStatut} \u2192 ${newStatut}`,
   816→          auteur: 'Utilisateur',
   817→          type: newStatut === 'Annulee' ? 'annulation' : newStatut === 'Cloturee' ? 'cloture' : 'modification',
   818→        }],
   819→        ...(newStatut === 'Cloturee' ? { dateCloture: today } : {}),
   820→        ...(newStatut === 'Publiee' && !o.datePublication ? { datePublication: today } : {}),
   821→      };
   822→    }));
   823→
   824→    /* Mettre à jour le drawer si ouvert */
   825→    setDrawerOffre(prev => prev && prev.id === id ? { ...prev, statutOffre: newStatut } : prev);
   826→
   827→    /* Simuler un délai API de 300ms puis effacer le spinner */
   828→    setTimeout(() => {
   829→      setLoadingStatus({ id: null, statut: null });
   830→      setSnack({ msg: `Statut de ${offre.numero} mis \u00e0 jour vers '${newStatut}'`, severity: 'success' });
   831→    }, 300);
   832→  }, [data]);

   835→  const handleArchive = useCallback((offre) => {

   843→

   847→    const { type, offre } = confirmDlg;
   848→    const today = new Date().toLocaleDateString('fr-FR');
   849→
   850→    if (type === 'archive') {
   851→      setData(prev => prev.map(o => o.id === offre.id ? {

   854→        dateCloture: o.dateCloture || today,
   855→      } : o));
   856→      setDrawerOffre(prev => prev && prev.id === offre.id ? { ...prev, statutOffre: 'Cloturee' } : prev);
   857→      setSnack({ msg: `${offre.numero} archiv\u00e9e avec succ\u00e8s`, severity: 'info' });
   858→    } else if (type === 'delete') {
   859→      setData(prev => prev.filter(o => o.id !== offre.id));
   860→      if (drawerOffre?.id === offre.id) setDrawerOffre(null);
   861→      setSnack({ msg: `${offre.numero} supprim\u00e9e`, severity: 'warning' });
   862→    }
   863→
   864→    setConfirmDlg(null);
   865→  }, [confirmDlg, drawerOffre]);
   866→
   867→  const handleExport = useCallback(() => {
   868→    const cols = ['N\u00b0','Intitul\u00e9','D\u00e9partement','Type Poste','Type Contrat','Canal','Statut Offre','Date Publication','Date Cl\u00f4ture','Date Requise','Responsable','Priorit\u00e9','Budget Allou\u00e9','Salaire Min','Salaire Max','Candidatures Re\u00e7ues','Site'];
   869→    const rows = [cols.join(';')];
   870→    filtered.forEach(o => rows.push([o.numero, o.intitule, o.departement, o.typePoste, o.typeContrat, o.canalDiffusion, o.statutOffre, o.datePublication, o.dateCloture, o.dateRequise, o.responsable, o.priorite, o.budgetAlloue, o.salaireMin, o.salaireMax, o.nbCandidaturesRecues, o.site].map(v => `"${v}"`).join(';')));
   871→    const blob = new Blob(['\uFEFF' + rows.join('\n')], { type: 'text/csv;charset=utf-8' });
   872→    const url = URL.createObjectURL(blob);
   873→    const a = document.createElement('a');
   874→    a.href = url;
   875→    a.download = 'offres_emploi.csv';
   876→    a.click();
   877→    URL.revokeObjectURL(url);
   878→    setSnack({ msg: 'Export CSV t\u00e9l\u00e9charg\u00e9', severity: 'info' });
   879→  }, [filtered]);
   880→
   881→  return (
   882→    <Box>
   883→      <Typography variant="h5" fontWeight="bold">Offres d'Emploi</Typography>
   884→      <Typography variant="body2" color="text.secondary" sx={{ mb: 2 }}>{filtered.length} offre(s) d'emploi</Typography>
   885→
   886→      <Box sx={{ display: 'flex', gap: 1, mb: 2, alignItems: 'center' }}>
   887→        <Button variant="outlined" startIcon={<Download fontSize="small" />} onClick={handleExport}>Exporter CSV</Button>
   888→        <Button variant="contained" startIcon={<Add fontSize="small" />} onClick={() => setDlg(true)}>Nouvelle Offre</Button>
   889→      </Box>
   890→
   891→      {/* KPIs */}
   892→      <Box sx={{ display: 'flex', gap: 2, mb: 3, flexWrap: 'wrap' }}>
   893→        <KPICard titre="TOTAL OFFRES" valeur={filtered.length} sousTexte={`${filtered.length} offre(s) enregistr\u00e9e(s)`} />
   894→        <KPICard titre="PUBLI\u00c9ES" valeur={filtered.filter(o => o.statutOffre === 'Publiee').length} sousTexte={`${Math.round(filtered.filter(o => o.statutOffre === 'Publiee').length / Math.max(filtered.length, 1) * 100)}% du total`} />
   895→        <KPICard titre="CANDIDATURES EN COURS" valeur={filtered.filter(o => o.statutOffre === 'Candidatures en cours').length} sousTexte="offres actives" />
   896→        <KPICard titre="TOTAL CANDIDATURES" valeur={filtered.reduce((s, o) => s + (o.nbCandidaturesRecues || 0), 0)} sousTexte="candidatures re\u00e7ues" />
   897→      </Box>
   898→
   899→      {/* Filtres */}
   900→      <Box sx={{ display: 'flex', gap: 2, mb: 2, flexWrap: 'wrap' }}>
   901→        <FormControl size="small" sx={{ minWidth: 180 }}>
   902→          <Select value={filterDep} onChange={e => { setFilterDep(e.target.value); setPage(0); }} displayEmpty>
   903→            <MenuItem value="Tous">Tous les d\u00e9partements</MenuItem>
   904→            {nomenclatures.departement.map(d => <MenuItem key={d} value={d}>{d}</MenuItem>)}
   905→          </Select>
   906→        </FormControl>
   907→        <FormControl size="small" sx={{ minWidth: 170 }}>
   908→          <Select value={filterStatut} onChange={e => { setFilterStatut(e.target.value); setPage(0); }} displayEmpty>
   909→            <MenuItem value="Tous">Tous les statuts</MenuItem>
   910→            {nomenclatures.statut_offre.map(s => <MenuItem key={s} value={s}>{s}</MenuItem>)}
   911→          </Select>
   912→        </FormControl>
   913→        <FormControl size="small" sx={{ minWidth: 170 }}>
   914→          <Select value={filterCanal} onChange={e => { setFilterCanal(e.target.value); setPage(0); }} displayEmpty>
   915→            <MenuItem value="Tous">Tous les canaux</MenuItem>
   916→            {nomenclatures.canal_diffusion.map(c => <MenuItem key={c} value={c}>{c}</MenuItem>)}
   917→          </Select>
   918→        </FormControl>
   919→      </Box>
   920→
   921→      {/* Tableau */}
   922→      <Paper>
   923→        <TableContainer><Table size="small">
   924→          <TableHead><TableRow>
   925→            {tableCols.map(c => (
   926→              <TableCell key={c.key} sx={{ fontWeight: 'bold', bgcolor: '#f5f5f5', whiteSpace: 'nowrap', width: c.width }}>{c.label}</TableCell>
   927→            ))}
   928→            <TableCell sx={{ fontWeight: 'bold', bgcolor: '#f5f5f5', width: 230, textAlign: 'center' }}>Actions</TableCell>
   929→          </TableRow></TableHead>
   930→          <TableBody>
   931→            {filtered.slice(page * rpp, page * rpp + rpp).map(row => (
   932→              <TableRow key={row.id} hover sx={{ cursor: 'pointer' }} onClick={() => setDrawerOffre(row)}>
   933→                {tableCols.map(c => (
   934→                  <TableCell key={c.key} sx={{ whiteSpace: 'nowrap' }}>
   935→                    {c.chip ? <Chip label={row[c.key]} size="small" variant="outlined" /> :
   936→                     row[c.key]}
   937→                  </TableCell>
   938→                ))}
   939→                <TableCell>
   940→                  <Box sx={{ display: 'flex', alignItems: 'center', gap: 0.8, justifyContent: 'center' }} onClick={e => e.stopPropagation()}>

   943→                      onStatutChange={handleQuickStatutChange}

   949→                      offre={row}
   950→                      onView={o => setDrawerOffre(o)}
   951→                      onArchive={handleArchive}
   952→                      onDelete={handleDelete}
   953→                    />
   954→                  </Box>
   955→                </TableCell>
   956→              </TableRow>
   957→            ))}
   958→            {filtered.length === 0 && (
   959→              <TableRow><TableCell colSpan={tableCols.length + 1} align="center" sx={{ py: 4, color: 'text.secondary' }}>Aucune offre trouv\u00e9e</TableCell></TableRow>
   960→            )}
   961→          </TableBody>
   962→        </Table></TableContainer>
   963→        <TablePagination
   964→          component="div" count={filtered.length} page={page}
   965→          onPageChange={(e, p) => setPage(p)} rowsPerPage={rpp}
   966→          onRowsPerPageChange={e => { setRpp(parseInt(e.target.value, 10)); setPage(0); }}
   967→          rowsPerPageOptions={[5, 10, 25]} labelRowsPerPage="Lignes par page"
   968→        />
   969→      </Paper>
   970→
   971→      {/* Drawer d'\u00e9dition */}
   972→      <OffreDrawer
   973→        offre={drawerOffre}
   974→        open={Boolean(drawerOffre)}
   975→        onClose={() => setDrawerOffre(null)}
   976→        onSave={handleSaveOffre}
   977→      />
   978→
   979→      {/* Dialogue d'ajout */}
   980→      <AddDialog
   981→        open={dlg} onClose={() => setDlg(false)} title="Nouvelle Offre d'Emploi"
   982→        fields={[
   983→          { key: 'intitule', label: 'Intitul\u00e9 du Poste', required: true },
   984→          { key: 'departement', label: 'D\u00e9partement', type: 'select', options: nomenclatures.departement, required: true },
   985→          { key: 'typePoste', label: 'Type Poste', type: 'select', options: nomenclatures.type_poste, required: true },
   986→          { key: 'typeContrat', label: 'Type Contrat', type: 'select', options: nomenclatures.type_contrat, required: true },
   987→          { key: 'canalDiffusion', label: 'Canal de Diffusion', type: 'select', options: nomenclatures.canal_diffusion, required: true },
   988→          { key: 'priorite', label: 'Priorit\u00e9', type: 'select', options: nomenclatures.priorite, required: true },
   989→          { key: 'dateRequise', label: 'Date Requise', required: true },
   990→          { key: 'dateCloture', label: 'Date de Cl\u00f4ture' },
   991→          { key: 'site', label: 'Site', type: 'select', options: nomenclatures.site },
   992→          { key: 'responsable', label: 'Responsable', required: true },
   993→          { key: 'roleResponsable', label: 'R\u00f4le du Responsable', type: 'select', options: nomenclatures.role_responsable },
   994→          { key: 'budgetAlloue', label: 'Budget Allou\u00e9 (FCFA)', type: 'number' },
   995→          { key: 'salaireMin', label: 'Salaire Min (FCFA)', type: 'number' },
   996→          { key: 'salaireMax', label: 'Salaire Max (FCFA)', type: 'number' },
   997→          { key: 'description', label: 'Description', multiline: true },
   998→        ]}
   999→        onSubmit={(vals) => {
  1000→          const nid = data.length + 1;
  1001→          const today = new Date().toLocaleDateString('fr-FR');
  1002→          setData(prev => [...prev, {
  1003→            id: nid, numero: 'OF-2025-' + String(nid).padStart(3, '0'),
  1004→            statutOffre: 'A creer', datePublication: '',
  1005→            nbCandidatures: 0, nbCandidaturesRecues: 0,
  1006→            candidatsAssocies: [],
  1007→            historique: [{ date: today, evenement: 'Offre cr\u00e9\u00e9e', auteur: vals.responsable || 'Utilisateur', type: 'creation' }],
  1008→            site: vals.site || 'Siege (Douala)',
  1009→            ...vals,
  1010→          }]);
  1011→        }}
  1012→      />
  1013→
  1014→      <Snackbar open={Boolean(snack)} autoHideDuration={3000} onClose={() => setSnack(null)} anchorOrigin={{ vertical: 'bottom', horizontal: 'center' }}>
  1015→        {snack && <Alert onClose={() => setSnack(null)} severity={snack.severity} variant="filled" sx={{ borderRadius: 2 }}>{snack.msg}</Alert>}
  1016→      </Snackbar>
  1017→
  1018→      {/* ═══ DIALOGUE DE CONFIRMATION (Archive / Supprimer) ═══ */}
  1019→      <Dialog
  1020→        open={Boolean(confirmDlg)}
  1021→        onClose={() => setConfirmDlg(null)}
  1022→        maxWidth="xs"
  1023→        PaperProps={{ sx: { borderRadius: 3, p: 0.5 } }}
  1024→      >
  1025→        {confirmDlg && (
  1026→          <>
  1027→            <DialogTitle sx={{ display: 'flex', alignItems: 'center', gap: 1.5, pb: 1 }}>
  1028→              <Box sx={{
  1029→                width: 40, height: 40, borderRadius: '50%',
  1030→                display: 'flex', alignItems: 'center', justifyContent: 'center',
  1031→                bgcolor: confirmDlg.type === 'delete' ? 'rgba(211,47,47,0.1)' : 'rgba(245,127,23,0.1)',
  1032→              }}>
  1033→                {confirmDlg.type === 'delete'
  1034→                  ? <Delete sx={{ color: '#d32f2f', fontSize: 22 }} />
  1035→                  : <Archive sx={{ color: '#f57f17', fontSize: 22 }} />
  1036→                }
  1037→              </Box>
  1038→              <Box>
  1039→                <Typography variant="subtitle1" fontWeight="bold" sx={{ fontSize: '1rem' }}>
  1040→                  {confirmDlg.type === 'delete' ? 'Supprimer cette offre' : 'Archiver cette offre'}
  1041→                </Typography>
  1042→                <Typography variant="caption" color="text.secondary">
  1043→                  {confirmDlg.offre.numero} — {confirmDlg.offre.intitule}
  1044→                </Typography>
  1045→              </Box>
  1046→            </DialogTitle>
  1047→            <DialogContent>
  1048→              <Typography variant="body2" sx={{ color: 'text.secondary', lineHeight: 1.6 }}>
  1049→                {confirmDlg.type === 'delete'
  1050→                  ? `Cette action est irréversible. L'offre ${confirmDlg.offre.numero} et toutes ses données associées (candidatures, historique) seront définitivement supprimées.`

  1054→            </DialogContent>
  1055→            <DialogActions sx={{ px: 3, pb: 2 }}>
  1056→              <Button onClick={() => setConfirmDlg(null)} color="inherit" sx={{ textTransform: 'none' }}>

  1069→            </DialogActions>

  1072→      </Dialog>
  1073→    </Box>
  1074→  );
