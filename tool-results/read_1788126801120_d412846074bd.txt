   244→            <Box sx={{ position: 'relative', ml: 1.5, pl: 3, borderLeft: '2px solid #e0e0e0' }}>
   245→              {d.historique.map((h, i) => (
   246→                <Box key={i} sx={{ position: 'relative', pb: i < d.historique.length - 1 ? 2 : 0 }}>
   247→                  <Box sx={{ position: 'absolute', left: -25, top: 2, width: 12, height: 12, borderRadius: '50%', bgcolor: histColor[h.type] || '#9e9e9e', border: '2px solid white', boxShadow: '0 0 0 1px ' + (histColor[h.type] || '#9e9e') }} />
   248→                  <Typography variant="caption" color="text.secondary" sx={{ fontSize: '0.7rem' }}>{h.date}</Typography>
   249→                  <Typography variant="body2" fontWeight={500} sx={{ fontSize: '0.85rem' }}>{h.evenement}</Typography>
   250→                  <Typography variant="caption" color="text.secondary" sx={{ fontSize: '0.7rem' }}>par {h.auteur}</Typography>
   251→                </Box>
   252→              ))}
   253→            </Box>
   254→          ) : <Typography variant="body2" color="text.secondary" sx={{ textAlign: 'center', py: 2 }}>Aucun historique</Typography>}
   255→        </Paper>
   256→
   257→        {/* Notes */}
   258→        <Paper variant="outlined" sx={{ p: 2, mb: 2.5 }}>
   259→          <Box sx={{ display: 'flex', justifyContent: 'space-between', alignItems: 'center', mb: 1 }}>
   260→            <Box sx={{ display: 'flex', alignItems: 'center', gap: 0.5 }}><NoteAdd sx={{ fontSize: 16 }} /><Typography variant="subtitle2" fontWeight="bold">Notes</Typography></Box>
   261→            {!editingNote && <IconButton size="small" onClick={() => setEditingNote(true)}><Edit sx={{ fontSize: 16 }} /></IconButton>}
   262→          </Box>
   263→          {d.notes ? <Typography variant="body2" sx={{ mb: 1.5, color: 'text.secondary', bgcolor: '#f5f5f5', p: 1.5, borderRadius: 1, fontSize: '0.85rem' }}>{d.notes}</Typography> : !editingNote && <Typography variant="body2" color="text.secondary" sx={{ mb: 1.5 }}>Aucune note</Typography>}
   264→          {editingNote && (
   265→            <Box sx={{ display: 'flex', gap: 1, mt: 1 }}>
   266→              <TextField fullWidth size="small" placeholder="Ajouter une note..." value={newNote} onChange={e => setNewNote(e.target.value)} onKeyDown={e => e.key === 'Enter' && handleSaveNote()} multiline rows={2} sx={{ '& textarea': { fontSize: '0.85rem' } }} />
   267→              <Button variant="contained" size="small" onClick={handleSaveNote} disabled={!newNote.trim()} sx={{ minWidth: 0, px: 2 }}><Send sx={{ fontSize: 16 }} /></Button>
   268→            </Box>
   269→          )}
   270→        </Paper>
   271→      </Box>
   272→
   273→      {/* Footer actions */}
   274→      {!isClosed && (
   275→        <Box sx={{ p: 2, borderTop: '1px solid', borderColor: 'divider', bgcolor: 'background.paper', display: 'flex', gap: 1, flexWrap: 'wrap' }}>
   276→          <FormControl size="small" sx={{ minWidth: 160 }}>
   277→            <Select value={d.statut} onChange={e => onStatutChange(d.id, e.target.value)} sx={{ fontSize: '0.8rem' }}>
   278→              {nomenclatures.statut_demande.map(s => <MenuItem key={s} value={s}>{s}</MenuItem>)}
   279→            </Select>
   280→          </FormControl>
   281→          <Button variant="contained" size="small" startIcon={<CheckCircle sx={{ fontSize: 16 }} />} onClick={() => onStatutChange(d.id, 'Pourvue')} sx={{ bgcolor: '#2e7d32', '&:hover': { bgcolor: '#1b5e20' } }}>Marquer pourvue</Button>
   282→          <Button variant="outlined" size="small" startIcon={<Business sx={{ fontSize: 16 }} />} onClick={() => { onClose(); navigate('/cabinets'); }}>Cabinet</Button>
   283→        </Box>
   284→      )}
   285→    </Drawer>
   286→  );

   290→export default function Demandes() {
   291→  const [data, setData] = useState(initialData);
   292→  const [dlg, setDlg] = useState(false);
   293→  const [filterDep, setFilterDep] = useState('Tous');
   294→  const [filterStatut, setFilterStatut] = useState('Tous');
   295→  const [filterPrio, setFilterPrio] = useState('Tous');
   296→  const [page, setPage] = useState(0);
   297→  const [rpp, setRpp] = useState(10);
   298→  const [drawerDemande, setDrawerDemande] = useState(null);
   299→  const [snack, setSnack] = useState(null);
   300→
   301→  const filtered = useMemo(() => data.filter(d =>
   302→    (filterDep === 'Tous' || d.departement === filterDep) &&
   303→    (filterStatut === 'Tous' || d.statut === filterStatut) &&
   304→    (filterPrio === 'Tous' || d.priorite === filterPrio)
   305→  ), [data, filterDep, filterStatut, filterPrio]);
   306→
   307→  const budgetTotal = useMemo(() => filtered.reduce((s, d) => s + (d.budgetSalaire || 0), 0), [filtered]);
   308→
   309→  const handleStatutChange = useCallback((id, newStatut) => {
   310→    setData(prev => prev.map(d => {
   311→      if (d.id !== id) return d;
   312→      const today = new Date().toLocaleDateString('fr-FR');
   313→      const newHist = [...(d.historique || []), { date: today, evenement: `Statut changé : ${d.statut} → ${newStatut}`, auteur: 'Utilisateur', type: newStatut === 'Pourvue' ? 'pourvoi' : 'validation' }];
   314→      return { ...d, statut: newStatut, historique: newHist, ...(newStatut === 'Pourvue' ? { datePourvue: today } : {}) };
   315→    }));
   316→    setDrawerDemande(prev => prev && prev.id === id ? { ...prev, statut: newStatut } : prev);
   317→    setSnack({ msg: `Statut mis à jour : ${newStatut}`, severity: 'success' });
   318→  }, []);
   319→
   320→  const handleAddNote = useCallback((id, note) => {
   321→    setData(prev => prev.map(d => {
   322→      if (d.id !== id) return d;
   323→      const today = new Date().toLocaleDateString('fr-FR');
   324→      const newHist = [...(d.historique || []), { date: today, evenement: `Note ajoutée : ${note.substring(0, 50)}${note.length > 50 ? '...' : ''}`, auteur: 'Utilisateur', type: 'creation' }];
   325→      return { ...d, notes: (d.notes ? d.notes + '\n' : '') + `[${today}] ${note}`, historique: newHist };
   326→    }));
   327→    setDrawerDemande(prev => prev && prev.id === id ? { ...prev, notes: (prev.notes ? prev.notes + '\n' : '') + `[${new Date().toLocaleDateString('fr-FR')}] ${note}` } : prev);
   328→    setSnack({ msg: 'Note ajoutée', severity: 'info' });
   329→  }, []);
   330→
   331→  const handleExport = useCallback(() => {
   332→    const cols = ['N°','Date','Département','Poste','Type Poste','Type Contrat','Effectif','Motif','Date Besoin','Priorité','Statut','Date Pourvue','Responsable','Rôle','Cabinet','Budget Salaire','Coût Recrutement','Délai (j)','Site','Notes'];
   333→    const rows = [cols.join(';')];
   334→    filtered.forEach(d => rows.push([d.numero,d.dateDemande,d.departement,d.posteRecherche,d.typePoste,d.typeContrat,d.effectif,d.motif,d.dateBesoin,d.priorite,d.statut,d.datePourvue,d.responsableDemande,d.roleResponsable,d.cabinetAgence,d.budgetSalaire,d.coutRecrutement,d.delai,d.site,d.notes].map(v => `"${v}"`).join(';')));
   335→    const blob = new Blob(['\uFEFF' + rows.join('\n')], { type: 'text/csv;charset=utf-8' });
   336→    const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'demandes_recrutement.csv'; a.click(); URL.revokeObjectURL(url);
   337→    setSnack({ msg: 'Export CSV téléchargé', severity: 'info' });
   338→  }, [filtered]);
   339→
   340→  return (
   341→    <Box>
   342→      <Typography variant="h5" fontWeight="bold">Demandes de Recrutement</Typography>
   343→      <Typography variant="body2" color="text.secondary" sx={{ mb: 2 }}>{filtered.length} demande(s) de recrutement</Typography>
   344→      <Box sx={{ display: 'flex', gap: 1, mb: 2 }}>
   345→        <Button variant="outlined" startIcon={<Download fontSize="small" />} onClick={handleExport}>Exporter CSV</Button>
   346→        <Button variant="contained" startIcon={<Add fontSize="small" />} onClick={() => setDlg(true)}>Nouvelle Demande</Button>
   347→      </Box>
   348→      <Box sx={{ display: 'flex', gap: 2, mb: 3, flexWrap: 'wrap' }}>
   349→        <KPICard titre="TOTAL DEMANDES" valeur={filtered.length} sousTexte={`${filtered.length} demande(s) enregistrée(s)`} />
   350→        <KPICard titre="POURVUES" valeur={filtered.filter(d => d.statut === 'Pourvue').length} sousTexte={`${Math.round(filtered.filter(d => d.statut === 'Pourvue').length / Math.max(filtered.length, 1) * 100)}% du total`} />
   351→        <KPICard titre="EN COURS" valeur={filtered.filter(d => ['En cours', 'Validée', 'En attente'].includes(d.statut)).length} sousTexte="demandes actives" />
   352→        <KPICard titre="BUDGET TOTAL" valeur={formatFCFA(budgetTotal)} sousTexte="budget cumulé" />
   353→      </Box>
   354→      <Box sx={{ display: 'flex', gap: 2, mb: 2, flexWrap: 'wrap' }}>
   355→        <FormControl size="small" sx={{ minWidth: 180 }}><Select value={filterDep} onChange={e => { setFilterDep(e.target.value); setPage(0); }} displayEmpty><MenuItem value="Tous">Tous les départements</MenuItem>{nomenclatures.departement.map(d => <MenuItem key={d} value={d}>{d}</MenuItem>)}</Select></FormControl>
   356→        <FormControl size="small" sx={{ minWidth: 150 }}><Select value={filterStatut} onChange={e => { setFilterStatut(e.target.value); setPage(0); }} displayEmpty><MenuItem value="Tous">Tous les statuts</MenuItem>{nomenclatures.statut_demande.map(s => <MenuItem key={s} value={s}>{s}</MenuItem>)}</Select></FormControl>

   375→                <TableCell>

   377→                </TableCell>
   378→              </TableRow>
   379→            ))}
   380→            {filtered.length === 0 && <TableRow><TableCell colSpan={tableCols.length + 1} align="center" sx={{ py: 4, color: 'text.secondary' }}>Aucune demande trouvée</TableCell></TableRow>}
