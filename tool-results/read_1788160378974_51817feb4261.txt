   128→/* ═══ KANBAN BOARD ═══ */
   129→function KanbanBoard({ data, onStatutChange, onCardClick }) {
   130→  const columns = useMemo(() => {
   131→    const map = {};
   132→    KANBAN_COLS.forEach(c => { map[c.id] = []; });
   133→    data.forEach(d => { if (map[d.statut]) map[d.statut].push(d); });
   134→    return map;
   135→  }, [data]);
   136→
   137→  const handleDragEnd = useCallback((result) => {
   138→    const { destination, source, draggableId } = result;
   139→    if (!destination || destination.droppableId === source.droppableId) return;
   140→    const id = parseInt(draggableId, 10);
   141→    onStatutChange(id, destination.droppableId);
   142→  }, [onStatutChange]);
   143→
   144→  return (
   145→    <DragDropContext onDragEnd={handleDragEnd}>
   146→      <Box sx={{ display: 'flex', gap: 1.5, overflowX: 'auto', pb: 2, minHeight: 500 }}>
   147→        {KANBAN_COLS.map(col => {
   148→          const items = columns[col.id] || [];
   149→          return (
   150→            <Droppable key={col.id} droppableId={col.id}>
   151→              {(provided, snapshot) => (
   152→                <Paper ref={provided.innerRef} {...provided.droppableProps}
   153→                  variant="outlined"
   154→                  sx={{
   155→                    minWidth: 280, maxWidth: 320, flex: '1 1 280px',
   156→                    display: 'flex', flexDirection: 'column', maxHeight: 'calc(100vh - 320px)',
   157→                    borderTop: `3px solid ${col.color}`,
   158→                    bgcolor: snapshot.isDraggingOver ? `${col.bgcolor}` : 'background.paper',
   159→                    transition: 'background-color 0.2s',
   160→                  }}
   161→                >
   162→                  {/* Column header */}
   163→                  <Box sx={{ p: 1.5, pb: 1, display: 'flex', alignItems: 'center', gap: 1, borderBottom: '1px solid', borderColor: 'divider' }}>
   164→                    <Typography variant="subtitle2" fontWeight="bold" sx={{ color: col.color, flex: 1, fontSize: '0.85rem' }}>{col.label}</Typography>
   165→                    <Badge badgeContent={items.length} sx={{ '& .MuiBadge-badge': { bgcolor: col.color, color: 'white', fontWeight: 700, fontSize: '0.7rem' } }} />
   166→                  </Box>
   167→
   168→                  {/* Cards */}
   169→                  <Box sx={{ flex: 1, overflowY: 'auto', p: 1, display: 'flex', flexDirection: 'column', gap: 1 }}>
   170→                    {items.map((d, idx) => {
   171→                      const jours = Math.floor((new Date() - new Date(d.dateDemande.split('/').reverse().join('-'))) / 86400000);
   172→                      const delaiPct = d.delai > 0 ? Math.min(100, Math.round((jours / d.delai) * 100)) : 0;
   173→                      const delaiColor = delaiPct >= 100 ? '#d32f2f' : delaiPct >= 75 ? '#f57f17' : '#2e7d32';
   174→                      return (
   175→                        <Draggable key={d.id} draggableId={String(d.id)} index={idx}>
   176→                          {(prov, snap) => (
   177→                            <Paper ref={prov.innerRef} {...prov.draggableProps}
   178→                              onClick={() => onCardClick(d)}
   179→                              sx={{
   180→                                p: 1.5, cursor: snap.isDragging ? 'grabbing' : 'grab',
   181→                                borderLeft: `3px solid ${prioriteColor[d.priorite] === 'error' ? '#d32f2f' : prioriteColor[d.priorite] === 'warning' ? '#f57f17' : prioriteColor[d.priorite] === 'info' ? '#1976d2' : '#bdbdbd'}`,
   182→                                boxShadow: snap.isDragging ? 4 : 1,
   183→                                opacity: snap.isDragging ? 0.9 : 1,
   184→                                transform: snap.isDragging ? 'rotate(2deg)' : 'none',
   185→                                transition: 'box-shadow 0.2s, transform 0.2s, opacity 0.2s',
   186→                                '&:hover': { boxShadow: 3 },
   187→                              }}
   188→                            >
   189→                              <Box sx={{ display: 'flex', alignItems: 'flex-start', gap: 1 }}>
   190→                                <Box {...prov.dragHandleProps} sx={{ mt: 0.3, color: 'text.disabled', '&:hover': { color: 'text.primary' } }}><DragIndicator sx={{ fontSize: 18 }} /></Box>

   213→                    {items.length === 0 && <Typography variant="body2" color="text.disabled" sx={{ textAlign: 'center', py: 4, fontSize: '0.8rem' }}>Aucune demande</Typography>}

   216→                </Paper>
   217→              )}
