   224→                size="small"
   225→                color={options?.colorMap?.[v] || 'default'}
   226→                sx={{ fontWeight: 600, height: 26, fontSize: '0.78rem' }}

   230→            {(options?.list || []).map(opt => (

   246→        <TextField
   247→          size="small"
   248→          type="date"
   249→          fullWidth
   250→          value={value || ''}

   260→            transition: 'background-color 0.2s',

   294→  }

   304→        fullWidth

   306→        onChange={e => onChange(editKey, e.target.value)}

   310→        sx={{
   311→          '& textarea, & input': { fontSize: '0.85rem' },
   312→          bgcolor: editing === editKey ? '#fffde7' : 'transparent',
   313→          transition: 'background-color 0.2s',
   314→        }}
   315→      />
   316→    </Box>
   317→  );
   318→}

   320→/* ═══ DRAWER D'ÉDITION INTERACTIVE ═══ */
   321→function OffreDrawer({ offre, open, onClose, onSave }) {
   322→  const [form, setForm] = useState({});
   323→  const [editing, setEditing] = useState(null);
   324→  const [hasChanges, setHasChanges] = useState(false);
   325→
   326→  /* Synchroniser le formulaire quand l'offre change */
   327→  useEffect(() => {
   328→    if (offre) {
   329→      setForm({
   330→        numero: offre.numero,
   331→        intitule: offre.intitule,
   332→        departement: offre.departement,
   333→        dateRequise: offre.dateRequise,
   334→        datePublication: offre.datePublication,
   335→        dateCloture: offre.dateCloture,
   336→        priorite: offre.priorite,
   337→        statutOffre: offre.statutOffre,
   338→        responsable: offre.responsable,
   339→        roleResponsable: offre.roleResponsable,
   340→        typeContrat: offre.typeContrat,
   341→        typePoste: offre.typePoste,
   342→        canalDiffusion: offre.canalDiffusion,
   343→        budgetAlloue: offre.budgetAlloue,
   344→        salaireMin: offre.salaireMin,
   345→        salaireMax: offre.salaireMax,
   346→        description: offre.description,
   347→        site: offre.site,
   348→      });
   349→      setHasChanges(false);
   350→      setEditing(null);
   351→    }
   352→  }, [offre]);
   353→
   354→  if (!offre) return null;
   355→
   356→  const d = offre;
   357→
   358→  const handleChange = (key, val) => {
   359→    setForm(prev => ({ ...prev, [key]: val }));
   360→    setHasChanges(true);
   361→  };
   362→
   363→  const handleEdit = (key) => setEditing(key);
   364→
   365→  const handleSave = () => {
   366→    if (onSave && hasChanges) {
   367→      onSave(d.id, form);
   368→      setHasChanges(false);
   369→      setEditing(null);
   370→    }
   371→  };
   372→
   373→  const handleCancel = () => {
   374→    /* Restaurer les valeurs originales */
   375→    setForm({
   376→      numero: d.numero, intitule: d.intitule, departement: d.departement,
   377→      dateRequise: d.dateRequise, datePublication: d.datePublication, dateCloture: d.dateCloture,
   378→      priorite: d.priorite, statutOffre: d.statutOffre, responsable: d.responsable,
   379→      roleResponsable: d.roleResponsable, typeContrat: d.typeContrat, typePoste: d.typePoste,
